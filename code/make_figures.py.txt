import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from pathlib import Path

DATA = Path("data")
FIGS = Path("figs")
FIGS.mkdir(exist_ok=True)

plt.rcParams.update({
    "font.size": 11, "axes.titlesize": 12, "axes.labelsize": 11,
    "legend.fontsize": 10, "figure.dpi": 120,
    "pdf.fonttype": 42
})

COLORS = {
    "Hamilton": "#1f77b4", "Madison": "#d62728", "Jay": "#2ca02c",
    "Joint": "#9467bd", "Disputed": "#7f7f7f"
}

def main():
    papers = pd.read_csv(DATA / "papers.csv")
    lex = pd.read_csv(DATA / "papers_lexicon_clean.csv")
    
    # --- Fig 1: blind ---
    sb = np.load(DATA / "scores_blind.npy")[:, :2]
    fig, ax = plt.subplots(figsize=(7, 5.5))
    for a in ["Hamilton", "Madison", "Jay", "Joint", "Disputed"]:
        m = (papers.author == a).values
        ax.scatter(sb[m, 0], sb[m, 1], c=COLORS[a], label=a, alpha=0.75, edgecolors="white", linewidths=0.6)
    ax.axhline(0, color="grey", lw=0.4, zorder=0)
    ax.axvline(0, color="grey", lw=0.4, zorder=0)
    ax.set_xlabel("PC1 (23.3% variance)")
    ax.set_ylabel("PC2 (9.9% variance)")
    ax.legend(title="Author", frameon=False)
    fig.tight_layout()
    fig.savefig(FIGS / "fig1_blind.pdf")
    plt.close(fig)
    
    # --- Fig 2: loadings ---
    cols = [c for c in lex.columns if c.startswith("w_") and c != "w_number"]
    X = lex[cols].values.astype(float)
    X = X - X.mean(axis=0)
    _, S, Vt = np.linalg.svd(X, full_matrices=False)
    load = Vt[0].copy()
    if load[cols.index("w_the")] < 0: load = -load
    
    top = np.argsort(-np.abs(load))[:20]
    words = [cols[i][2:] for i in top]
    vals = load[top]
    colors = ["#d62728" if v > 0 else "#1f77b4" for v in vals]
    
    fig, ax = plt.subplots(figsize=(6.5, 6))
    y = np.arange(len(words))[::-1]
    ax.barh(y, vals, color=colors, edgecolor="white", height=0.75)
    ax.set_yticks(y)
    ax.set_yticklabels(words)
    ax.axvline(0, color="black", lw=0.6)
    ax.set_xlim(-0.9, 0.9)
    ax.set_xlabel("Loading on PC1 (blind, mean-centred)")
    fig.tight_layout()
    fig.savefig(FIGS / "fig2_loadings.pdf")
    plt.close(fig)

    # --- Fig 3: calibrated ---
    sc = np.load(DATA / "scores_calibrated.npy")[:, :2]
    fig, ax = plt.subplots(figsize=(7, 5.5))
    for a in ["Hamilton", "Madison"]:
        m = (papers.author == a).values
        ax.scatter(sc[m, 0], sc[m, 1], c=COLORS[a], label=a, alpha=0.8, edgecolors="white", linewidths=0.6)
    ax.axhline(0, color="grey", lw=0.4, zorder=0)
    ax.axvline(0, color="grey", lw=0.4, zorder=0)
    ax.set_xlabel("PC1 (67.0% variance)")
    ax.set_ylabel("PC2 (11.4% variance)")
    ax.legend(title="Author", frameon=False)
    fig.tight_layout()
    fig.savefig(FIGS / "fig3_calibrated.pdf")
    plt.close(fig)

    # --- Fig 4: spec curve ---
    try:
        sv = np.load(DATA / "spec_verdicts.npy")
        nums = np.load(DATA / "spec_disputed_nums.npy")
        is_mad = (sv == "Madison")
        frac = is_mad.mean(axis=0) * 100
        order = np.argsort(-frac)
        names = [f"Fed {int(n)}" for n in nums[order]]
        vals = frac[order]
        cols = ["#d62728" if int(n) == 55 else "#1f77b4" for n in nums[order]]
        
        fig, ax = plt.subplots(figsize=(6.5, 5))
        y = np.arange(len(names))[::-1]
        ax.barh(y, vals, color=cols, edgecolor="white", height=0.75)
        ax.set_yticks(y)
        ax.set_yticklabels(names)
        ax.set_xlim(0, 105)
        ax.set_xlabel("% of 500 specifications assigning to Madison")
        ax.axvline(90, color="grey", ls="--", lw=0.6)
        fig.tight_layout()
        fig.savefig(FIGS / "fig4_spec_curve.pdf")
        plt.close(fig)
    except FileNotFoundError:
        print("spec_verdicts.npy not found, skipping Fig 4")

    print("Figures saved to figs/")

if __name__ == "__main__":
    main()
